\begin{frame}{Le problème \qui{Yahya}}
  \begin{columns}[T]

    \begin{block}{Un client écrit à sa banque}
      \og Je veux acheter un appartement. \fg\\

      \og Pourquoi j'ai payé 8\eur{} de frais ce mois-ci ? \fg
    \end{block}
    \medskip
    L'assistant doit \textbf{comprendre}, \textbf{poser les bonnes questions}, tenir compte de
    \textbf{la situation du client} et proposer une piste.
    \column{0.44\textwidth}
    \begin{itemize}
      \item Un Français sur quatre est client de La Banque Postale.
      \item L'expérience client est sa \textbf{priorité IA n\textsuperscript{o}\,1}.
      \item La règle : \textbf{open source, local}, le plus petit modèle qui suffit,
            frugalité \textbf{mesurée}.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Le plus petit moyen qui suffit, message par message \qui{Yahya}}
  \small
  \begin{tabularx}{\textwidth}{@{}l Y l@{}}
    \toprule
    \textbf{Niveau} & \textbf{Quand, et comment} & \textbf{Coût} \\
    \midrule
    \textcolor{mainblue}{\textbf{0 --- règles}} & Demande claire : mots-clés ou classifieur ($\geq \tau$), parcours YAML,
      règles publiques, gabarits. Plusieurs demandes : l'urgence d'abord. & 0 LLM \\
    \addlinespace
    \textcolor{orange!80!black}{\textbf{1 --- comprendre}} & Message flou, réponse libre, renvoi au contexte : un
      \textbf{petit LLM} + \textbf{RAG d'exemples} rend un \textbf{JSON imposé par une grammaire}, vérifié par le
      code. Il ne répond jamais au client. & 1 appel \\
    \addlinespace
    \textcolor{violet}{\textbf{2 --- répondre}} & Question ouverte : \textbf{RAG sur 37 fiches sourcées}, un LLM
      rédige en citant ; garde-fou (chiffres, fidélité), sinon l'extrait. & 1 appel \\
    \bottomrule
  \end{tabularx}
  \medskip

  La décision est prise \textbf{à chaque message} : une conversation passe de 1 à 2, puis à 0. La mémoire
  (parcours, demandes en file, informations données) est commune ; après une parenthèse, on reprend.
  \medskip

  \centering\textit{\og Le vélo pour aller chercher le pain, la voiture pour le week-end. \fg}
\end{frame}

\begin{frame}{Explicabilité : pourquoi, et pourquoi pas \qui{Sohayb}}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Léa} : 250\,000\eur{}, 25\,000\eur{} d'apport, 2\,900\eur{} nets.
    \begin{itemize}
      \item Mensualité $\approx$ 1\,197\eur{} sur 25 ans à 3,32\,\% (taux moyen Banque de France,
            daté), hors assurance.
      \item \textcolor{kored}{\textbf{41,3\,\% $>$ 35\,\%}} : règle \f{HCSF-taux-effort} ; non retenu,
            sans refus bancaire.
      \item Écarté : prêt sur 30 ans (\f{HCSF-duree}).
      \item Budget indicatif $\approx$ 215\,000\eur{} ; à 200\,000\eur{}, 32,2\,\% : un conseiller étudie.
    \end{itemize}
    \column{0.46\textwidth}
    \begin{block}{Ce que voit le client}
      Les règles appliquées, les produits proposés \emph{et écartés}, la source de chaque
      paramètre. Le calcul se vérifie à la main.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Confiance et conformité \qui{Sohayb}}
  \begin{itemize}
    \item \textbf{Règlement européen sur l'IA} : la notation de solvabilité est à haut risque ;
          nous n'en faisons pas, un conseiller décide. Le client sait qu'il parle à une IA.
    \item \textbf{Mention légale} sur toute réponse de crédit ; aucune promesse.
    \item \textbf{Aucun accès au compte, aucune opération} (périmètre du kick-off) : l'assistant
          explique et oriente ; rappel : jamais de code confidentiel.
    \item \textbf{LLM encadrés} : au niveau 1, il ne voit pas le profil et ne répond pas ; au niveau 2, il ne
          répond qu'à partir de fiches sourcées, sinon \og je n'ai pas cette information \fg.
    \item \textbf{Données simulées, traitement local} : rien ne sort de la machine.
    \item \textbf{Inclusion} : mots simples, jargon expliqué, offre clientèle fragile.
  \end{itemize}
\end{frame}
